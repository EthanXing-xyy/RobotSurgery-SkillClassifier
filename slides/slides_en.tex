\begin{frame}{Why automate skill assessment?}
\begin{itemize}
\item Surgical skill predicts outcomes: the lowest-rated quartile of surgeons had far more complications than the highest-rated one (Birkmeyer et al., 2013).
\item Expert rating with OSATS is slow, costly and partly subjective.
\item A surgical robot already logs every instrument movement and the endoscope video, so assessment can use data that are collected anyway.
\item A useful system must work for a \alert{surgeon it has never seen}, so we hold out whole surgeons (LOUO), not single trials.
\end{itemize}
\end{frame}

\begin{frame}{Task and data}
\small JIGSAWS Suturing: \NTrials{} trials from 8 surgeons; label = self-reported level, N (4 surgeons, 19 trials), I (2, 10), E (2, 10). 76 kinematic channels at 30\,Hz and endoscope video. Official LOUO folds hold out one surgeon, hence one class. Only \alert{two surgeons per trained class}.
\vspace{4pt}
\fig[0.66\textheight]{fig2_data.pdf}
\end{frame}

\begin{frame}{Pipeline}
\fig[0.60\textheight]{fig1_pipeline.pdf}
\vspace{2pt}
{\small Every block $\rightarrow$ standardise ($\rightarrow$ PCA for V-appear) $\rightarrow$ L2 logistic regression; $C$ chosen by an inner LOUO loop on the training surgeons only. Late fusion = mean of class probabilities.}
\end{frame}

\begin{frame}{Features}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Kinematics (patient-side instruments)}
\begin{itemize}
\item \textbf{K-simple (30)}: path length, speed, angular speed, idle fraction, economy, gripper statistics, duration, bimanual coordination.
\item \textbf{K-texture (10)}, \alert{focused improvement}: approximate entropy, log dimensionless jerk, spectral entropy. Idea from Zia \& Essa: experts move more regularly.
\end{itemize}
\column{0.5\textwidth}
\textbf{Video (left camera)}
\begin{itemize}
\item \textbf{V-appear}: frozen ResNet-18 at 1\,Hz, mean + SD of embeddings, in-fold PCA.
\item \textbf{V-dyn (8)}: how fast consecutive embeddings change; cancels static appearance of a session.
\item \textbf{V-flow (15)}: Farneback optical flow at 2\,Hz.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Evaluation protocol}
\begin{itemize}
\item Metric: \textbf{pooled macro-F1} over the 39 out-of-fold predictions (per-fold macro-F1 is undefined with one class per fold).
\item Uncertainty: \textbf{user-level bootstrap} (2000 resamples of surgeons) and an \textbf{exact permutation test} over all $8!/(4!\,2!\,2!) = 420$ label assignments.
\item Pre-registered in \texttt{PROTOCOL.md}: all models and three comparisons fixed before the run; every model evaluated once.
\item \textbf{Shortcut audit}: surgeon-identity probe per block; duration-only classifier; features residualised on log duration.
\end{itemize}
\end{frame}

\begin{frame}{Where it fails: the experts}
\fig[0.42\textheight]{fig4_confusion.pdf}
\begin{itemize}
\item Novices F1 \FNMOne{}, intermediates \FIMOne{}, experts only \FEMOne{}: 5 of 10 expert trials called N, 4 called I.
\item Holding out one expert leaves \alert{one} expert in training, so the model learns that person rather than expertise.
\item Experts work at the pace of intermediates (median \DurE{}\,s vs \DurI{}\,s; novices \DurN{}\,s).
\end{itemize}
\end{frame}

\begin{frame}{Shortcut audit: identity and pace}
\fig[0.52\textheight]{fig6_audit.pdf}
\begin{itemize}\small
\item Surgeon identity from one trial: K-simple \IdKSimple{}, V-appear \IdVAppear{} (chance 0.125).
\item Duration alone reaches \FCOne{}; removing log duration from K-simple drops it from \FMOne{} to \FCTwo{}.
\item Texture failed: log jerk vs log duration $|r| = \PaceKTextureMax{}$, identity \IdKTexture{}. V-dyn mostly repeats pace ($|r|$ \PaceVDyn{}).
\end{itemize}
\end{frame}

\begin{frame}{How this differs from prior work}
\begin{itemize}
\item Strict unseen-surgeon evaluation with macro-F1 and explicit uncertainty; many published near-100\% results use LOSO.
\item Zia \& Essa's 88.2\% LOUO accuracy cannot be reproduced as a pooled count over 39 trials, so it is not directly comparable.
\item We \textbf{measure} identity and pace shortcuts instead of assuming a high score means skill.
\item Pre-registration and an exact permutation test make negative results credible.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
\item Simple kinematic statistics transfer to new surgeons (\FMOne{} vs \FMZero{}), mostly through \alert{pace}.
\item Whole-frame video and simple fusion did not help; motion texture did not help.
\item Experts cannot be learned from one training expert.
\item Next: more surgeons (all JIGSAWS tasks, other datasets), pace-independent features, instrument-region video, fusion weights learned in-fold.
\end{itemize}
